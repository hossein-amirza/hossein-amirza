"use strict"; document.addEventListener("DOMContentLoaded", () => { const menuButton = document.querySelector(".menu-toggle"); const mainNav = document.querySelector("#main-nav"); // منوی موبایل if (menuButton && mainNav) { const closeMenu = () => { menuButton.setAttribute("aria-expanded", "false"); mainNav.classList.remove("is-open"); }; menuButton.addEventListener("click", () => { const isOpen = menuButton.getAttribute("aria-expanded") === "true"; menuButton.setAttribute( "aria-expanded", String(!isOpen) ); mainNav.classList.toggle("is-open", !isOpen); }); // بستن منو پس از انتخاب یک لینک mainNav.querySelectorAll("a").forEach((link) => { link.addEventListener("click", closeMenu); }); // بستن منو با کلید Escape document.addEventListener("keydown", (event) => { if (event.key === "Escape") { closeMenu(); } }); // بستن منو با کلیک خارج از آن document.addEventListener("click", (event) => { if ( !mainNav.contains(event.target) && !menuButton.contains(event.target) ) { closeMenu(); } }); // هنگام بازگشت به نمایش دسکتاپ، منو بسته شود window.addEventListener("resize", () => { if (window.innerWidth > 820) { closeMenu(); } }); } // سال جاری در پابرگ const year = document.querySelector("#year"); if (year) { year.textContent = new Intl.NumberFormat("fa-IR") .format(new Date().getFullYear()); } // آماده‌سازی ایمیل از فرم تماس const form = document.querySelector("#contact-form"); const note = document.querySelector("#form-note"); if (form) { form.addEventListener("submit", (event) => { event.preventDefault(); const name = document.querySelector("#name").value.trim(); const email = document.querySelector("#email").value.trim(); const message = document.querySelector("#message").value.trim(); if (!name || !email || !message) { if (note) { note.textContent = "لطفاً نام، ایمیل و متن پیام را تکمیل کنید."; } return; } const subject = encodeURIComponent( "پیام از وب‌سایت — " + name ); const body = encodeURIComponent( "نام: " + name + "\nایمیل: " + email + "\n\nمتن پیام:\n" + message ); const mailto = "mailto:hosseinmohammadi.writer@gmail.com" + "?subject=" + subject + "&body=" + body; if (note) { note.textContent = "در حال باز کردن برنامهٔ ایمیل…"; } window.location.href = mailto; }); } });
